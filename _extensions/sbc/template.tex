\documentclass[12pt]{article}

% Quarto/Pandoc adapter for the SBC LaTeX style.
% Source of truth for layout: sbc-template.sty supplied with the SBC template.
\usepackage{sbc-template}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{amsmath,amssymb}
\usepackage{url}
$if(natbib)$
\usepackage{natbib}
% The supplied sbc.bst is a copy of apalike whose documented SBC-specific
% difference is the removal of the comma before the year in citation labels.
% Keep apalike as the executable bibliography formatter and reproduce that
% punctuation explicitly in natbib while the original sbc.bst remains tracked
% by provenance hash.
\setcitestyle{authoryear,round,aysep={ }}
$endif$
\usepackage{hyperref}
$if(natbib)$
% sbc-template.sty installs a historical hyperref-specific \@lbibitem hook at
% the end of the preamble. That hook predates natbib and overwrites natbib's
% author-year label parser. Preserve the natbib+hyperref definition here and
% restore it after the SBC hook has run, keeping the vendored style untouched.
\makeatletter
\let\quartosbcnatbiblbibitem\@lbibitem
\AtEndPreamble{\let\@lbibitem\quartosbcnatbiblbibitem}
\makeatother
$endif$

\sloppy

$if(title)$
\title{$title$}
$endif$

$if(by-author)$
\author{$for(by-author)$$by-author.name.literal$$if(by-author.metadata.institute)$\inst{$by-author.metadata.institute$}$endif$$sep$, $endfor$}
$else$
$if(author)$
\author{$for(author)$$author$$sep$, $endfor$}
$endif$
$endif$

$if(sbc-affiliations)$
\address{$for(sbc-affiliations)$$sbc-affiliations.text$$if(sbc-affiliations.email)$\email{$sbc-affiliations.email$}$endif$$sep$
\nextinstitute
$endfor$}
$else$
\address{}
$endif$

$endif$

$endif$

$body$

$if(natbib)$

$endif$
$endif$

\end{document}
